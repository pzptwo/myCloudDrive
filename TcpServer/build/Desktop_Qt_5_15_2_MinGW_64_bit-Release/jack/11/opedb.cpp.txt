int OpeDB::handleSearchUsr(const char *name)
{
    if(NULL==name)
    {
        return -1;//不存在
    }
    QString data=QString("select online from usrInfo where name =\'%1\'").arg(name);//替换的形式,
   // qDebug()<<data;
    QSqlQuery query;
    query.exec(data);
    //用int类型，是因为有三种结果，
    if(query.next())
    {
       int ret= query.value(0).toInt();
        if(1==ret){
           return 1;//在线
        }
        else if(0==ret){
            return 0;//不在线
        }
    }
    else
    {
        return -1;
    }
}

int OpeDB::handleAddFriendCheck(const char *pername, const char *name)
{
    if(NULL==pername||NULL==name)
    {
        return -1;
    }

    //链表查询，连表查询，自身，这里是数据库的知识。，这里拼接字符串
    QString data=QString("select * from friend where "
                           "(id=(select id from usrInfo where name=\'%1\') and "  // and前加空格
                           "friendId=(select id from usrInfo where name=\'%2\')) or "  // or前加空格
                           "(id=(select id from usrInfo where name=\'%3\') and "  // and前加空格
                           "friendId=(select id from usrInfo where name=\'%4\'))")
                       .arg(pername).arg(name).arg(name).arg(pername);

    qDebug()<<data;
    QSqlQuery query;
    query.exec(data);
    if(query.next())
    {
        return 0;//双方已是好友
        }

    else
    {   //为啥不直接调用
        data=QString("select online from usrInfo where name =\'%1\'").arg(pername);//替换的形式
        qDebug()<<data;
        QSqlQuery query;
        query.exec(data);
        //用int类型，是因为有三种结果，
        if(query.next())
        {
            int ret= query.value(0).toInt();
            if(1==ret){
                return 1;//在线
            }
            else if(0==ret){
                return 2;//不在线
            }
        }
        else {
            return 3;//不存在
        }
    }
}
